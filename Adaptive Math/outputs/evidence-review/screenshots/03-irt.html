1// Math-only learning models. Default item parameters are provisional, not calibrated.
2export const LEVELS = ['REMEMBERING', 'UNDERSTANDING', 'APPLYING', 'ADVANCED_APPLYING'];
3const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
4export function itemParameters(q: any) {
5 return {
6 a: Number.isFinite(q.irtA) && q.irtA > 0 ? q.irtA : 1.2,
7 b: Number.isFinite(q.irtB) ? q.irtB : [-1.5, -0.5, 0.5, 1.5][Math.max(0, LEVELS.indexOf(q.cognitiveLevel))],
8 c: Number.isFinite(q.irtC) && q.irtC >= 0 && q.irtC < 1 ? q.irtC : 0.25
9 };
10}
11export function probability(theta: number, q: any) {
12 const { a, b, c } = itemParameters(q);
13 const z = a * (theta - b);
14 const logistic = z >= 0 ? 1 / (1 + Math.exp(-z)) : Math.exp(z) / (1 + Math.exp(z));
15 return clamp(c + (1 - c) * logistic, 1e-9, 1 - 1e-9);
16}
17export function information(theta: number, q: any) {
18 const p = probability(theta, q), { a, c } = itemParameters(q);
19 return a * a * (1 - p) * (p - c) ** 2 / ((1 - c) ** 2 * p);
20}
21export function estimateAbility(answers: any[]) {
22 let theta = 0;
23 for (let i = 0; i < 60; i++) {
24 let score = -theta / 4, info = 0.25;
25 for (const answer of answers) {
26 const q = answer.question, p = probability(theta, q), { a, c } = itemParameters(q);
27 score += a * (Number(answer.isCorrect) - p) * (p - c) / (p * (1 - c));
28 info += information(theta, q);
29 }
30 const step = clamp(score / info, -0.75, 0.75);
31 theta = clamp(theta + step, -4, 4);
32 if (Math.abs(step) < 1e-5) break;
33 }
34 return { theta, standardError: 1 / Math.sqrt(0.25 + answers.reduce((sum, a) => sum + information(theta, a.question), 0)) };
35}
