55 if (a.isCorrect) { levelIndex++; failures = 0; }

57 }
58 if (stopped || levelIndex >= availableLevels.length) continue;
59 // An unanswered verification item is required before a gap can be confirmed.
60 const items = pool.filter(q => q.cognitiveLevel === availableLevels[levelIndex] && !used.has(q.id));
61 if (!items.length) continue;
62 items.sort((a, b) => information(theta, b) - information(theta, a) || tie(seed, a.id) - tie(seed, b.id));
63 candidates.push({ question: items[0], count: history.length });
64 }
65 candidates.sort((a, b) => a.count - b.count || tie(seed, a.question.skillId) - tie(seed, b.question.skillId));
66 return candidates[0]?.question ?? null;
